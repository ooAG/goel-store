import { create } from 'zustand';
import { persist } from 'zustand/middleware';

export const useCart = create()(
  persist(
    (set, get: any) => ({
      items: [],
      addItem: (product: any) => {
        const items = get().items;
        const existing = items.find((i: any) => i.id === product.id);
        if (existing) {
          set({ items: items.map((i: any) => i.id === product.id ? { ...i, quantity: i.quantity + 1 } : i) });
        } else {
          set({ items: [...items, { ...product, quantity: 1 }] });
        }
      },
      removeItem: (id: string) => {
        const items = get().items;
        const existing = items.find((i: any) => i.id === id);
        if (existing?.quantity === 1) {
          set({ items: items.filter((i: any) => i.id !== id) });
        } else {
          set({ items: items.map((i: any) => i.id === id ? { ...i, quantity: i.quantity - 1 } : i) });
        }
      },
      clearCart: () => set({ items: [] }),
      getTotal: () => get().items.reduce((acc: number, item: any) => acc + (item.sellingPrice * item.quantity), 0),
    }),
    { name: 'goel-store-cart' }
  )
);

export async function POST(req: Request) {
  try {
    const body = await req.json();
    let customer = await prisma.customer.findUnique({ where: { phone: body.phone } });
    if (!customer) customer = await prisma.customer.create({ data: { name: body.name, phone: body.phone, address: body.address, dues: 0 } });

    const months = ["JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV", "DEC"];

    const prefix = `${months[d.getMonth()]}${String(d.getFullYear()).slice(-2)}`;
    const chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789";
    let customId = "";
    while (true) {

      for (let i = 0; i < 3; i++) r += chars.charAt(Math.floor(Math.random() * chars.length));

    if (newDues > 0) {
        storeCreditUsed = Math.min(newDues, cartTotal);
        newDues -= storeCreditUsed;
        await prisma.ledger.create({

        });
    }

    const remainingTotal = cartTotal - storeCreditUsed;

    if (remainingTotal > 0 && body.paymentMode === "UDHAAR") {
        newDues -= remainingTotal;
        await prisma.ledger.create({

        });

    if (newDues !== customer.dues) {
        await prisma.customer.update({
            where: { id: customer.id },
            data: { dues: newDues }
        });
    }

    const order = await prisma.order.create({
      data: { 
        id: customId, customerId: customer.id, 
        originalAmount: cartTotal,
        totalAmount: remainingTotal, 
        paymentMode: body.paymentMode || "CASH", 
        assignedQr: body.assignedQr || "CASH / MANUAL",
        paymentStatus: body.paymentMode === "UDHAAR" ? "DUES_ADDED" : (remainingTotal === 0 ? "PAID" : "PENDING"),

export async function PATCH(req: Request) {
  try {
    const body = await req.json();
    const order = await prisma.order.findUnique({ where: { id: body.id } });
    if (!order) return NextResponse.json({ error: "Not found" }, { status: 404 });

    if (body.action === "REVISE_ORDER") {
       const newItems = body.items || []; 
       const newOriginalAmount = newItems.reduce((sum: number, item: any) => sum + (item.sellingPrice * item.quantity), 0);
       const originalAmt = order.originalAmount || 0;
       const diff = originalAmt - newOriginalAmount; 
       
       const isNowEmpty = newOriginalAmount === 0;

       if (diff > 0) {
           const reason = isNowEmpty 
              ? `Order #${order.id} Full Return. (Total Bill ₹${originalAmt} refunded to Khata/Credit)`
              : `Order #${order.id} Revised. (Old Bill: ₹${originalAmt}, New Bill: ₹${newOriginalAmount}, Refunded: ₹${diff})`;
           
           if (order.paymentStatus === "DUES_ADDED" || order.paymentMode === "UDHAAR" || order.paymentStatus === "PAID" || (order.originalAmount || 0) > (order.totalAmount || 0)) {
               await prisma.customer.update({ where: { id: order.customerId }, data: { dues: { increment: diff } } });
               await prisma.ledger.create({ data: { customerId: order.customerId, amount: diff, reason } });
           }
       } else if (diff < 0) {
           const extraCharge = Math.abs(diff);
           if (order.paymentStatus === "DUES_ADDED" || order.paymentMode === "UDHAAR") {
               await prisma.customer.update({ where: { id: order.customerId }, data: { dues: { decrement: extraCharge } } });
               await prisma.ledger.create({ data: { customerId: order.customerId, amount: -extraCharge, reason: `Order #${order.id} Revised. Bill increased by ₹${extraCharge}.` } });
           }
       }
       
       await prisma.orderItem.deleteMany({ where: { orderId: order.id } });
       const newTotalAmount = Math.max(0, (order.totalAmount || 0) - diff);

       return NextResponse.json(await prisma.order.update({
           where: { id: order.id },
           data: { 
               originalAmount: newOriginalAmount, 
               totalAmount: newTotalAmount,
               orderStatus: isNowEmpty ? "CANCELLED" : order.orderStatus,
               paymentStatus: isNowEmpty ? "CANCELLED" : (newTotalAmount === 0 ? "PAID" : order.paymentStatus),
               items: { create: newItems.map((item: any) => ({ productId: item.productId || null, customName: item.customName || null, quantity: item.quantity, sellingPrice: item.sellingPrice })) }
           }
       }));
    }

    if (body.action === "CONVERT_TO_COD") { 
        await prisma.order.update({ where: { id: body.id }, data: { paymentMode: "CASH", assignedQr: "CASH / COD" } }); 
        return NextResponse.json({ success: true }); 
    }

    if (body.action === "CHANGE_QR") { 
        await prisma.order.update({ where: { id: body.id }, data: { assignedQr: body.assignedQr } }); 
        return NextResponse.json({ success: true }); 
    }

    // 🔥 THE DUAL-LOG ENTRY FIX FOR PARTIAL SETTLEMENT PAYMENT ROUTE 🔥
    if (body.action === "SETTLE_PAYMENT") {
       if (order.paymentStatus === "PAID") return NextResponse.json({ error: "Already Settled" }, { status: 400 });

       const amountPaid = parseFloat(body.amountReceived);
       const orderTotal = order.totalAmount || 0;
       const netDiff = amountPaid - orderTotal;

       if (order.paymentStatus === "DUES_ADDED") {
           await prisma.customer.update({ where: { id: order.customerId }, data: { dues: { increment: amountPaid } } });
       } else if (amountPaid !== orderTotal) {
           await prisma.customer.update({ where: { id: order.customerId }, data: { dues: { increment: netDiff } } });
       }

       // 1. Log positive statement entry for the exact cash piece received safely
       await prisma.ledger.create({ 
           data: { customerId: order.customerId, amount: amountPaid, reason: `Order #${order.id} Payment Received. (Cash/UPI)` } 
       });

       // 2. If short-fall exist, log an automated separate transparent minus entry right away!
       if (netDiff < 0) {
           await prisma.ledger.create({
               data: { customerId: order.customerId, amount: netDiff, reason: `Order #${order.id} Shortfall. (Remaining ₹${Math.abs(netDiff)} moved to Udhaar)` }
           });
       }
       // 3. If excess tip/advance cash exist, log an automated separate positive credit entry!
       else if (netDiff > 0) {
           await prisma.ledger.create({
               data: { customerId: order.customerId, amount: netDiff, reason: `Order #${order.id} Advance. (Excess ₹${netDiff} added as Store Credit)` }
           });
       }

       return NextResponse.json(await prisma.order.update({ where: { id: order.id }, data: { paymentStatus: "PAID" } }));
    }

       if (order.paymentStatus === "DUES_ADDED") return NextResponse.json({ error: "Already in Khata" }, { status: 400 });

       if (order.orderStatus === "CANCELLED") return NextResponse.json({ error: "Already Cancelled" }, { status: 400 });
